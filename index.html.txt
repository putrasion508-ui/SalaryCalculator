  function fmtPlain(x) { const v = round2(x); return Number.isInteger(v) ? String(v) : v.toFixed(2).replace('.', ','); }

  function parseAmount(raw) {
    if (raw == null) return { ok: false, empty: true, error: 'Wajib diisi' };
    let s = String(raw).trim();
    s = s.replace(/^usd\s*/i, '').replace(/\s*usd$/i, '').replace(/^\$\s*/, '').replace(/[\s ']/g, '');
    if (s === '') return { ok: false, empty: true, error: 'Wajib diisi' };
    const bad = { ok: false, error: 'Format angka tidak valid' };
    if (s.charAt(0) === '-' || s.charAt(0) === '−') return { ok: false, error: 'Tidak boleh negatif' };
    if (!/^[0-9.,]+$/.test(s)) return bad;
    const lastDot = s.lastIndexOf('.'), lastComma = s.lastIndexOf(',');
    let norm;
    if (lastDot >= 0 && lastComma >= 0) {
      const dec = lastDot > lastComma ? '.' : ',';
      const grp = dec === '.' ? ',' : '.';
      const parts = s.split(dec);
      if (parts.length !== 2) return bad;
      const intPart = parts[0], frac = parts[1];
      if (!/^\d+$/.test(frac)) return bad;
      const re = grp === '.' ? /^\d{1,3}(\.\d{3})*$/ : /^\d{1,3}(,\d{3})*$/;
      if (!re.test(intPart)) return bad;
      norm = intPart.split(grp).join('') + '.' + frac;
    } else if (lastDot >= 0 || lastComma >= 0) {
      const sep = lastDot >= 0 ? '.' : ',';
      const parts = s.split(sep);
      if (parts.length > 2) {
        const re = sep === '.' ? /^[1-9]\d{0,2}(\.\d{3})+$/ : /^[1-9]\d{0,2}(,\d{3})+$/;
        if (!re.test(s)) return bad;
        norm = parts.join('');
      } else {
        const a = parts[0], b = parts[1];
        if (a === '' && b === '') return bad;
        if (b === '') norm = a;
        else if (/^[1-9]\d{0,2}$/.test(a) && /^\d{3}$/.test(b)) norm = a + b;
        else norm = (a || '0') + '.' + b;
      }
    } else norm = s;
    if (!/^\d+(\.\d+)?$/.test(norm)) return bad;
    const v = Number(norm);
    if (!isFinite(v)) return bad;
    return { ok: true, value: round2(v) };
  }
  function readAmount(raw, kind) {
    const r = parseAmount(raw);
    if (r.empty) {
      if (kind === 'ded') return { ok: true, value: 0 };
      return { ok: false, error: kind === 'annual' ? 'Annual salary wajib diisi' : 'Nominal wajib diisi' };
    }
    if (!r.ok) return r;
    const v = r.value;
    if (kind === 'annual') {
      if (v <= 0) return { ok: false, error: 'Harus lebih dari 0' };
      if (v > LIMITS.annualMax) return { ok: false, error: 'Maksimal $' + fmtNum(LIMITS.annualMax, 0) };
    } else if (kind === 'ded') {
      if (v > LIMITS.dedMax) return { ok: false, error: 'Maksimal $' + fmtNum(LIMITS.dedMax, 0) };
    } else {
      if (v <= 0) return { ok: false, error: 'Harus lebih dari 0' };
      if (v > LIMITS.claimMax) return { ok: false, error: 'Maksimal $' + fmtNum(LIMITS.claimMax, 0) };
    }
    return { ok: true, value: v };
  }

  // ---------- state ----------
  function emptyState() { return { version: 1, months: {}, days: {}, claims: {}, history: {}, ui: { period: null, brush: 'ON', rangeTon: 0, rangeToff: 0 } }; }
  function newId() { return 'c' + Date.now().toString(36) + Math.random().toString(36).slice(2, 8); }
  function isObj(o) { return o !== null && typeof o === 'object' && !Array.isArray(o); }
  function numOk(v, min, max) { return typeof v === 'number' && isFinite(v) && v >= min && v <= max; }
  function cleanCfg(c) {
    if (!isObj(c)) return null;
    if (!numOk(c.annual, 0, LIMITS.annualMax) || !numOk(c.insurance, 0, LIMITS.dedMax) || !numOk(c.pension, 0, LIMITS.dedMax)) return null;
    return { annual: round2(c.annual), insurance: round2(c.insurance), pension: round2(c.pension) };
  }
  function cleanClaims(arr, counter) {
    const out = [];
    if (!Array.isArray(arr)) { if (arr !== undefined) counter.n++; return out; }
    const seen = new Set();
    for (const c of arr) {
      if (out.length >= LIMITS.claimsPerPeriod) { counter.n++; continue; }
      if (!isObj(c) || !numOk(c.amount, 0.01, LIMITS.claimMax)) { counter.n++; continue; }
      let id = (typeof c.id === 'string' && /^[A-Za-z0-9_-]{1,40}$/.test(c.id) && !seen.has(c.id)) ? c.id : newId();
      while (seen.has(id)) id = newId();
      seen.add(id);
      const desc = typeof c.desc === 'string' ? c.desc.replace(/[\u0000-\u001f\u007f]/g, ' ').trim().slice(0, LIMITS.descMax) : '';
      out.push({ id: id, desc: desc || 'Claim expenses', amount: round2(c.amount) });
    }
    return out;
  }
  function cleanDateList(arr, period, counter) {
    const out = [];
    if (!Array.isArray(arr)) { if (arr !== undefined) counter.n++; return out; }
    const seen = new Set();
    for (const d of arr) {
      if (typeof d === 'string' && parseDate(d) && d.slice(0, 7) === period && !seen.has(d)) { seen.add(d); out.push(d); }
      else counter.n++;
    }
    return out.sort();
  }
  function sanitizeState(raw) {
    const st = emptyState();
    const counter = { n: 0 };
    if (!isObj(raw)) return { state: st, dropped: raw == null ? 0 : 1 };
    if (isObj(raw.months)) for (const k of Object.keys(raw.months)) {
      const c = cleanCfg(raw.months[k]);
      if (parsePeriod(k) && c) st.months[k] = c; else counter.n++;
    }
    if (isObj(raw.days)) for (const k of Object.keys(raw.days)) {
      let v = raw.days[k];
      if (v === 'TRV') v = 'TON'; // data versi lama: satu jenis "Travel"
      if (parseDate(k) && (v === 'ON' || v === 'TON' || v === 'TOFF')) st.days[k] = v; else counter.n++;
    }
    if (isObj(raw.claims)) for (const k of Object.keys(raw.claims)) {
      if (!parsePeriod(k)) { counter.n++; continue; }
      const list = cleanClaims(raw.claims[k], counter);
      if (list.length) st.claims[k] = list;
    }
    if (isObj(raw.history)) for (const k of Object.keys(raw.history)) {
      const h = raw.history[k];
      if (!parsePeriod(k) || !isObj(h)) { counter.n++; continue; }
      const cfg = cleanCfg(h);
      if (!cfg || !(cfg.annual > 0)) { counter.n++; continue; }
      const on = cleanDateList(h.on, k, counter);
      const used = new Set(on);
      const ton = cleanDateList(h.ton !== undefined ? h.ton : h.trv, k, counter).filter(function (d) { return !used.has(d); });
      ton.forEach(function (d) { used.add(d); });
      const toff = cleanDateList(h.toff, k, counter).filter(function (d) { return !used.has(d); });
      st.history[k] = {
        period: k,
        savedAt: (typeof h.savedAt === 'string' && !isNaN(Date.parse(h.savedAt))) ? h.savedAt : new Date(0).toISOString(),
        annual: cfg.annual, insurance: cfg.insurance, pension: cfg.pension,
        on: on, ton: ton, toff: toff, claims: cleanClaims(h.claims, counter)
      };
    }
    if (isObj(raw.ui)) {
      if (parsePeriod(raw.ui.period)) st.ui.period = raw.ui.period;
      const b = raw.ui.brush === 'TRV' ? 'TON' : raw.ui.brush;
      if (BRUSHES.indexOf(b) >= 0) st.ui.brush = b;
      if (isInt(raw.ui.rangeTon, 0, LIMITS.travelMax)) st.ui.rangeTon = raw.ui.rangeTon;
      if (isInt(raw.ui.rangeToff, 0, LIMITS.travelMax)) st.ui.rangeToff = raw.ui.rangeToff;
    }
    return { state: st, dropped: counter.n };
  }

  function cfgFor(state, period) {
    const own = state.months[period];
    if (own) return { annual: own.annual, insurance: own.insurance, pension: own.pension, from: period, inherited: false };
    let best = null, firstLater = null;
    for (const k of Object.keys(state.months)) {
      if (k < period) { if (!best || k > best) best = k; }
      else if (!firstLater || k < firstLater) firstLater = k;
    }
    const src = best || firstLater;
    if (src) { const c = state.months[src]; return { annual: c.annual, insurance: c.insurance, pension: c.pension, from: src, inherited: true }; }
    return { annual: 0, insurance: 0, pension: 0, from: null, inherited: true };
  }
  function daysOf(dayMap, period) {
    const q = parsePeriod(period);
    const on = [], ton = [], toff = [];
    const n = daysInMonth(q.y, q.m);
    for (let d = 1; d <= n; d++) {
      const k = ymd(q.y, q.m, d);
      const v = dayMap[k];
      if (v === 'ON') on.push(k); else if (v === 'TON') ton.push(k); else if (v === 'TOFF') toff.push(k);
    }
    return { on: on, ton: ton, toff: toff };
  }
  function liveInputs(state, period) {
    const cfg = cfgFor(state, period);
    const d = daysOf(state.days, period);
    const claims = (state.claims[period] || []).map(function (c) { return { id: c.id, desc: c.desc, amount: c.amount }; });
    return { period: period, annual: cfg.annual, insurance: cfg.insurance, pension: cfg.pension, on: d.on, ton: d.ton, toff: d.toff, claims: claims };
  }
  function calc(inp) {
    const q = parsePeriod(inp.period);
    const days = daysInMonth(q.y, q.m);
    const onDates = inp.on || [], tonDates = inp.ton || [], toffDates = inp.toff || [];
    const on = onDates.length, ton = tonDates.length, toff = toffDates.length, trv = ton + toff, off = days - on - trv;
    const annual = inp.annual;
    const dr = annual / 365;
    const onAdd = (RATES.ON - RATES.OFF) * dr;
    const trvAdd = (RATES.TRV - RATES.OFF) * dr;
    const guarantee = round2(RATES.OFF * annual / 12);
    const onPay = round2(on * onAdd);
    const tonPay = round2(ton * trvAdd);
    const toffPay = round2(toff * trvAdd);
    const trvPay = round2(tonPay + toffPay);
    const gross = round2(guarantee + onPay + tonPay + toffPay);
    const insurance = round2(inp.insurance || 0), pension = round2(inp.pension || 0);
    const claims = (inp.claims || []).map(function (c) { return { id: c.id, desc: c.desc, amount: round2(c.amount) }; });
    const claimsTotal = round2(claims.reduce(function (s, c) { return s + c.amount; }, 0));
    const deductions = round2(insurance + pension);
    const net = round2(gross - deductions + claimsTotal);
    return {
      period: inp.period, payPeriod: addMonths(inp.period, 1), days: days, on: on, ton: ton, toff: toff, trv: trv, off: off,
      onDates: onDates.slice(), tonDates: tonDates.slice(), toffDates: toffDates.slice(), annual: annual, dr: dr, onAdd: onAdd, trvAdd: trvAdd,
      guarantee: guarantee, onPay: onPay, tonPay: tonPay, toffPay: toffPay, trvPay: trvPay, gross: gross, insurance: insurance, pension: pension,
      deductions: deductions, claims: claims, claimsTotal: claimsTotal, net: net
    };
  }
  function signature(inp) {
    return JSON.stringify([round2(inp.annual), round2(inp.insurance), round2(inp.pension), inp.on || [], inp.ton || [], inp.toff || [],
      (inp.claims || []).map(function (c) { return [c.desc, round2(c.amount)]; })]);
  }
  function status(state, period) {
    const h = state.history[period];
    if (!h) return 'none';
    return signature(h) === signature(liveInputs(state, period)) ? 'saved' : 'changed';
  }
  function makeSnapshot(inp, savedAt) {
    return {
      period: inp.period, savedAt: savedAt, annual: round2(inp.annual), insurance: round2(inp.insurance), pension: round2(inp.pension),
      on: inp.on.slice(), ton: inp.ton.slice(), toff: inp.toff.slice(),
      claims: inp.claims.map(function (c) { return { id: c.id, desc: c.desc, amount: round2(c.amount) }; })
    };
  }
  function clearPeriod(dayMap, period) {
    const q = parsePeriod(period);
    let n = 0;
    const dim = daysInMonth(q.y, q.m);
    for (let d = 1; d <= dim; d++) { const k = ymd(q.y, q.m, d); if (dayMap[k]) { delete dayMap[k]; n++; } }
    return n;
  }
  function restoreSnapshot(state, period) {
    const h = state.history[period];
    if (!h) return false;
    clearPeriod(state.days, period);
    h.on.forEach(function (d) { state.days[d] = 'ON'; });
    h.ton.forEach(function (d) { state.days[d] = 'TON'; });
    h.toff.forEach(function (d) { state.days[d] = 'TOFF'; });
    state.months[period] = { annual: h.annual, insurance: h.insurance, pension: h.pension };
    if (h.claims.length) state.claims[period] = h.claims.map(function (c) { return { id: c.id, desc: c.desc, amount: c.amount }; });
    else delete state.claims[period];
    return true;
  }
  function rangeInfo(from, to, type, opt) {
    if (!parseDate(from) || !parseDate(to)) return { ok: false, error: 'Tanggal belum lengkap atau tidak valid.' };
    if (from > to) return { ok: false, error: 'Tanggal "Dari" harus sebelum atau sama dengan "Sampai".' };
    if (BRUSHES.indexOf(type) < 0) return { ok: false, error: 'Jenis tidak valid.' };
    const n = diffDays(from, to) + 1;
    if (n > LIMITS.rangeMaxDays) return { ok: false, error: 'Rentang maksimal ' + LIMITS.rangeMaxDays + ' hari.' };
    let on = 0, ton = 0, toff = 0, off = 0;
    if (type === 'ON') {
      const a = (opt && opt.tonDays != null) ? opt.tonDays : 0;
      const b = (opt && opt.toffDays != null) ? opt.toffDays : 0;
      if (!isInt(a, 0, LIMITS.travelMax) || !isInt(b, 0, LIMITS.travelMax)) return { ok: false, error: 'Jumlah hari travel harus 0–' + LIMITS.travelMax + '.' };
      if (a + b > n) return { ok: false, error: 'Hari travel (' + (a + b) + ') lebih banyak dari panjang rentang (' + n + ' hari).' };
      ton = a; toff = b; on = n - a - b;
    } else if (type === 'TON') ton = n;
    else if (type === 'TOFF') toff = n;
    else off = n;
    return { ok: true, n: n, on: on, ton: ton, toff: toff, off: off, periods: periodsBetween(from.slice(0, 7), to.slice(0, 7)) };
  }
  // Rentang ON: N hari pertama = Travel ON, M hari terakhir = Travel OFF, sisanya ON
  function applyRange(dayMap, from, to, type, opt) {
    const info = rangeInfo(from, to, type, opt);
    if (!info.ok) return info;
    let d = from;
    for (let i = 0; i < info.n; i++) {
      let v = type;
      if (type === 'ON') v = i < info.ton ? 'TON' : (i >= info.n - info.toff ? 'TOFF' : 'ON');
      if (v === 'OFF') delete dayMap[d]; else dayMap[d] = v;
      d = addDays(d, 1);
    }
    return info;
  }
  function yearCount(dayMap, y) {
    const pre = y + '-';
    let on = 0, ton = 0, toff = 0;
    for (const k of Object.keys(dayMap)) {
      if (k.slice(0, 5) !== pre) continue;
      const v = dayMap[k];
      if (v === 'ON') on++; else if (v === 'TON') ton++; else if (v === 'TOFF') toff++;
    }
    return { on: on, ton: ton, toff: toff, trv: ton + toff };
  }
  function dayRanges(dates) {
    const ds = dates.map(function (d) { return +d.slice(8, 10); }).sort(function (a, b) { return a - b; });
    const parts = [];
    let i = 0;
    while (i < ds.length) {
      let j = i;
      while (j + 1 < ds.length && ds[j + 1] === ds[j] + 1) j++;
      parts.push(i === j ? String(ds[i]) : ds[i] + '–' + ds[j]);
      i = j + 1;
    }
    return parts.join(', ');
  }
  function rotationBlocks(dayMap) {
    const keys = Object.keys(dayMap).filter(function (k) { const v = dayMap[k]; return v === 'ON' || v === 'TON' || v === 'TOFF'; }).sort();
    const out = [];
    let cur = null;
    for (const k of keys) {
      if (cur && diffDays(cur.end, k) === 1) { cur.end = k; cur.days++; }
      else { cur = { start: k, end: k, days: 1, on: 0, ton: 0, toff: 0 }; out.push(cur); }
      const v = dayMap[k];
      if (v === 'ON') cur.on++; else if (v === 'TON') cur.ton++; else cur.toff++;
    }
    return out;
  }
  function historyItems(state) {
    return Object.keys(state.history).sort().reverse().map(function (p) {
      return { period: p, snap: state.history[p], res: calc(state.history[p]), status: status(state, p) };
    });
  }
  function historyYears(state, group) {
    const ys = new Set();
    for (const p of Object.keys(state.history)) ys.add(+((group === 'pay' ? addMonths(p, 1) : p).slice(0, 4)));
    return Array.from(ys).sort(function (a, b) { return b - a; });
  }
  function totals(results) {
    const t = { count: results.length, on: 0, ton: 0, toff: 0, trv: 0, off: 0, gross: 0, insurance: 0, pension: 0, deductions: 0, claims: 0, net: 0 };
    for (const r of results) {
      t.on += r.on; t.ton += r.ton; t.toff += r.toff; t.trv += r.trv; t.off += r.off;
      t.gross += r.gross; t.insurance += r.insurance; t.pension += r.pension; t.deductions += r.deductions; t.claims += r.claimsTotal; t.net += r.net;
    }
    ['gross', 'insurance', 'pension', 'deductions', 'claims', 'net'].forEach(function (k) { t[k] = round2(t[k]); });
    return t;
  }
  function historySummary(state, year, group) {
    const items = historyItems(state).filter(function (it) {
      return +((group === 'pay' ? it.res.payPeriod : it.period).slice(0, 4)) === year;
    });
    const t = totals(items.map(function (it) { return it.res; }));
    t.items = items;
    return t;
  }

  return {
    MONTHS: MONTHS, MONTHS_S: MONTHS_S, RATES: RATES, EXPECTED_ON: EXPECTED_ON, LIMITS: LIMITS, BRUSHES: BRUSHES, TYPE_LABEL: TYPE_LABEL, isInt: isInt,
    pad2: pad2, ymd: ymd, pkey: pkey, daysInMonth: daysInMonth, parsePeriod: parsePeriod, parseDate: parseDate,
    addMonths: addMonths, periodsBetween: periodsBetween, addDays: addDays, diffDays: diffDays, weekdayMon0: weekdayMon0,
    periodLabel: periodLabel, periodShort: periodShort, dateLabel: dateLabel,
    round2: round2, fmtNum: fmtNum, fmtMoney: fmtMoney, fmtInput: fmtInput, fmtPlain: fmtPlain, parseAmount: parseAmount, readAmount: readAmount,
    emptyState: emptyState, newId: newId, sanitizeState: sanitizeState, cfgFor: cfgFor, daysOf: daysOf, liveInputs: liveInputs, calc: calc,
    signature: signature, status: status, makeSnapshot: makeSnapshot, clearPeriod: clearPeriod, restoreSnapshot: restoreSnapshot,
    rangeInfo: rangeInfo, applyRange: applyRange, yearCount: yearCount, dayRanges: dayRanges, rotationBlocks: rotationBlocks,
    historyItems: historyItems, historyYears: historyYears, historySummary: historySummary, totals: totals
  };
})();

    return o;
  }
  function rgb(hex) {
    const n = parseInt(String(hex).slice(1), 16) || 0;
    return [(n >> 16) & 255, (n >> 8) & 255, n & 255].map(function (v) { return (Math.round(v / 255 * 1000) / 1000).toString(); }).join(' ');
  }
  function num(v) { return (Math.round(v * 100) / 100).toString(); }
  function wrap(str, maxW, size, bold) {
    const words = String(str).split(/\s+/).filter(Boolean);
    const lines = [];
    let cur = '';
    for (let w of words) {
      while (w && width(w, size, bold) > maxW) {
        let k = w.length;
        while (k > 1 && width(w.slice(0, k), size, bold) > maxW) k--;
        if (cur) { lines.push(cur); cur = ''; }
        lines.push(w.slice(0, k));
        w = w.slice(k);
      }
      if (!w) continue;
      const t = cur ? cur + ' ' + w : w;
      if (width(t, size, bold) <= maxW) cur = t; else { if (cur) lines.push(cur); cur = w; }
    }
    if (cur) lines.push(cur);
    return lines.length ? lines : [''];
  }

  function Doc() { this.W = 595.28; this.H = 841.89; this.pages = []; this.cur = null; }
  Doc.prototype.addPage = function () { this.cur = []; this.pages.push(this.cur); };
  Doc.prototype.text = function (x, y, str, o) {
    o = o || {};
    const size = o.size || 10, bold = !!o.bold;
    const w = width(str, size, bold);
    let tx = x;
    if (o.align === 'right') tx = x - w; else if (o.align === 'center') tx = x - w / 2;
    this.cur.push('BT /' + (bold ? 'F2' : 'F1') + ' ' + num(size) + ' Tf ' + rgb(o.color || '#000000') + ' rg ' + num(tx) + ' ' + num(this.H - y) + ' Td (' + enc(str) + ') Tj ET');
    return w;
  };
  Doc.prototype.rect = function (x, y, w, h, fill) { this.cur.push(rgb(fill) + ' rg ' + num(x) + ' ' + num(this.H - y - h) + ' ' + num(w) + ' ' + num(h) + ' re f'); };
  Doc.prototype.line = function (x1, y1, x2, y2, color, lw) { this.cur.push(rgb(color || '#000000') + ' RG ' + num(lw || 0.5) + ' w ' + num(x1) + ' ' + num(this.H - y1) + ' m ' + num(x2) + ' ' + num(this.H - y2) + ' l S'); };
  Doc.prototype.onPage = function (i, fn) { const save = this.cur; this.cur = this.pages[i]; fn(); this.cur = save; };
  Doc.prototype.output = function (meta) {
    const objs = [];
    const n = this.pages.length;
    const kids = [];
    for (let i = 0; i < n; i++) kids.push((6 + 2 * i) + ' 0 R');
    objs.push('<< /Type /Catalog /Pages 2 0 R >>');
    objs.push('<< /Type /Pages /Kids [' + kids.join(' ') + '] /Count ' + n + ' >>');
    objs.push('<< /Type /Font /Subtype /Type1 /BaseFont /Helvetica /Encoding /WinAnsiEncoding >>');
    objs.push('<< /Type /Font /Subtype /Type1 /BaseFont /Helvetica-Bold /Encoding /WinAnsiEncoding >>');
    objs.push('<< /Title (' + enc(meta.title || '') + ') /Producer (Gaji Rotasi) /Creator (Gaji Rotasi) /CreationDate (D:' + meta.date + ') >>');
    for (let i = 0; i < n; i++) {
      objs.push('<< /Type /Page /Parent 2 0 R /MediaBox [0 0 ' + num(this.W) + ' ' + num(this.H) + '] /Resources << /Font << /F1 3 0 R /F2 4 0 R >> >> /Contents ' + (7 + 2 * i) + ' 0 R >>');
      objs.push({ stream: this.pages[i].join('\n') });
    }
    let out = '%PDF-1.4\n%âãÏÓ\n';
    const offs = [];
    for (let i = 0; i < objs.length; i++) {
      offs.push(out.length);
      const o = objs[i];
      out += (i + 1) + ' 0 obj\n';
      if (typeof o === 'string') out += o + '\nendobj\n';
      else out += '<< /Length ' + o.stream.length + ' >>\nstream\n' + o.stream + '\nendstream\nendobj\n';
    }
    const xref = out.length;
    out += 'xref\n0 ' + (objs.length + 1) + '\n0000000000 65535 f \n';
    for (const off of offs) out += ('0000000000' + off).slice(-10) + ' 00000 n \n';

  function Layout() {
    this.doc = new Doc();
    this.W = this.doc.W; this.H = this.doc.H;
    this.m = 40; this.top = 40; this.bottom = 52;
    this.cw = this.W - 2 * this.m;
    this.pageHeader = null;
    this.newPage();
  }
  Layout.prototype.newPage = function () { this.doc.addPage(); this.y = this.top; if (this.pageHeader) this.pageHeader(this); };
  Layout.prototype.ensure = function (h) { if (this.y + h > this.H - this.bottom) { this.newPage(); return true; } return false; };
  Layout.prototype.text = function (str, o) {
    o = o || {};
    const size = o.size || 10, lh = o.lh || size * 1.4;
    const lines = wrap(str, (o.width || this.cw) - (o.indent || 0), size, o.bold);
    for (const ln of lines) {
      this.ensure(lh);
      this.y += lh;
      this.doc.text(this.m + (o.indent || 0), this.y - lh / 2 + size * 0.35, ln, { size: size, bold: o.bold, color: o.color });
    }
  };
  Layout.prototype.table = function (cols, rows, o) {
    o = o || {};
    const size = o.size || 8.5, pad = 4, lh = size * 1.3;
    const sum = cols.reduce(function (s, c) { return s + c.w; }, 0);
    const ws = cols.map((c) => c.w * this.cw / sum);
    const self = this;
    function cellX(i, x, align) { return align === 'r' ? x + ws[i] - pad : align === 'c' ? x + ws[i] / 2 : x + pad; }
    function drawHeader() {
      const lines = cols.map(function (c, i) { return wrap(c.label, ws[i] - pad * 2, size, true); });
      const nl = Math.max.apply(null, lines.map(function (l) { return l.length; }));
      const h = nl * lh + pad * 2;
      self.doc.rect(self.m, self.y, self.cw, h, o.headFill || '#0b2a3c');
      let x = self.m;
      lines.forEach(function (ls, i) {
        ls.forEach(function (ln, k) {
          self.doc.text(cellX(i, x, cols[i].align), self.y + pad + k * lh + lh / 2 + size * 0.35, ln,
            { size: size, bold: true, color: o.headColor || '#ffffff', align: cols[i].align === 'r' ? 'right' : cols[i].align === 'c' ? 'center' : 'left' });
        });
        x += ws[i];
      });
      self.y += h;
    }
    this.ensure(lh * 3 + pad * 4);
    drawHeader();
    rows.forEach(function (r, ri) {
      const cells = Array.isArray(r) ? r : r.cells;
      const bold = !Array.isArray(r) && !!r.bold;
      const fill = !Array.isArray(r) && r.fill;
      const lines = cells.map(function (t, i) { return wrap(t == null ? '' : String(t), ws[i] - pad * 2, size, bold); });
      const nl = Math.max.apply(null, lines.map(function (l) { return l.length; }));
      const h = nl * lh + pad * 2;
      if (self.y + h > self.H - self.bottom) { self.newPage(); drawHeader(); }
      if (fill) self.doc.rect(self.m, self.y, self.cw, h, fill);
      else if (ri % 2 === 1 && o.zebra !== false) self.doc.rect(self.m, self.y, self.cw, h, '#f3f7f9');
      let x = self.m;
      lines.forEach(function (ls, i) {
        ls.forEach(function (ln, k) {
          self.doc.text(cellX(i, x, cols[i].align), self.y + pad + k * lh + lh / 2 + size * 0.35, ln,
            { size: size, bold: bold, color: '#14232d', align: cols[i].align === 'r' ? 'right' : cols[i].align === 'c' ? 'center' : 'left' });
        });
        x += ws[i];
      });
      self.y += h;
      self.doc.line(self.m, self.y, self.m + self.cw, self.y, '#d5dee4', 0.4);
    });
  };
  return { Doc: Doc, Layout: Layout, width: width, wrap: wrap, enc: enc };
})();

  function loadFromRaw(raw, quiet) {
    if (raw == null) { S = C.emptyState(); return; }
    let parsed;
    try { parsed = JSON.parse(raw); }
    catch (e) {
      try { localStorage.setItem(KEY + '.corrupt', raw); } catch (e2) { /* ignore */ }
      S = C.emptyState();
      if (!quiet) toast('Data tersimpan rusak dan tidak bisa dibaca. Salinannya disisihkan; aplikasi mulai dari data kosong.', 'err', 7000);
      return;
    }
    const r = C.sanitizeState(parsed);
    S = r.state;
    if (r.dropped > 0 && !quiet) toast(r.dropped + ' entri data tidak valid diabaikan.', 'warn', 5000);
  }
  function load() {
    let raw = null;
    try { raw = localStorage.getItem(KEY); }
    catch (e) { toast('Penyimpanan perangkat tidak bisa diakses. Data hanya bertahan selama aplikasi terbuka.', 'err', 7000); return; }
    loadFromRaw(raw, false);
  }
  function persist() {
    try { localStorage.setItem(KEY, JSON.stringify(S)); return true; }
    catch (e) { toast('Gagal menyimpan ke perangkat (penyimpanan penuh atau diblokir). Perubahan belum tersimpan permanen.', 'err', 6000); return false; }
  }

  let modal = null;
  let clickShieldUntil = 0;
  function isModalOpen() { return !!modal; }
  // Setelah modal ditutup, klik kedua dari double-tap jangan "tembus" ke tombol di bawahnya.
  document.addEventListener('click', function (e) {
    if (Date.now() < clickShieldUntil && !(e.target && e.target.closest && e.target.closest('.modal'))) {
      e.preventDefault();
      e.stopPropagation();
    }
  }, true);
  function openModal(o) {
    if (modal) return Promise.resolve(null);
    return new Promise(function (resolve) {
      const root = $('modalRoot');
      root.innerHTML = '<div class="modal-back"><div class="modal" role="dialog" aria-modal="true" aria-labelledby="modalTitle"><h3 id="modalTitle"></h3><div class="modal-body"></div><div class="modal-btns"></div></div></div>';
      const back = root.firstChild;
      back.querySelector('#modalTitle').textContent = o.title;
      const body = back.querySelector('.modal-body');
      if (typeof o.body === 'string') { const p = document.createElement('p'); p.textContent = o.body; body.appendChild(p); }
      else if (o.body) body.appendChild(o.body);
      const btns = back.querySelector('.modal-btns');
      let done = false;
      // tap kedua dari double-tap yang membuka modal tidak boleh langsung menutup/mengonfirmasi
      const openedAt = Date.now();
      const tooSoon = function () { return Date.now() - openedAt < 350; };
      function close(v) {
        if (done) return;
        done = true;
        modal = null;
        clickShieldUntil = Date.now() + 400;
        document.removeEventListener('keydown', onKey);
        document.body.classList.remove('modal-open');
        root.innerHTML = '';
        resolve(v);
      }
      function onKey(e) { if (e.key === 'Escape') close(null); }
      o.buttons.forEach(function (b) {
        const el = document.createElement('button');
        el.type = 'button';
        el.className = 'btn' + (b.kind ? ' ' + b.kind : '');
        el.textContent = b.label;
        el.addEventListener('click', function () {
          if (done || tooSoon()) return;
          if (o.onButton && o.onButton(b.value) === false) return;
          close(b.value);
        });
        btns.appendChild(el);
      });
      back.addEventListener('click', function (e) { if (e.target === back && !tooSoon()) close(null); });
      document.addEventListener('keydown', onKey);
      document.body.classList.add('modal-open');
      modal = { close: close };
      const focusEl = body.querySelector('input,select') || btns.lastChild;
      if (focusEl && !('ontouchstart' in window)) focusEl.focus();
    });
  }
  function confirmModal(title, text, okLabel, danger) {
    return openModal({ title: title, body: text, buttons: [{ label: 'Batal', value: false }, { label: okLabel, value: true, kind: danger ? 'danger' : 'primary' }] })
      .then(function (v) { return v === true; });
  }

  /* ----- downloads ----- */
  function downloadBlob(blob, name) {
    const url = URL.createObjectURL(blob);
    const a = document.createElement('a');
    a.href = url; a.download = name; a.rel = 'noopener'; a.style.display = 'none';
    document.body.appendChild(a);
    a.click();
    a.remove();
    setTimeout(function () { URL.revokeObjectURL(url); }, 60000);
  }
  function canShareFiles() {
    try { return !!(navigator.canShare && window.File && navigator.canShare({ files: [new File(['x'], 'x.pdf', { type: 'application/pdf' })] })); }
    catch (e) { return false; }
  }
  function shareOrDownload(blob, name) {
    let file;
    try { file = new File([blob], name, { type: blob.type }); } catch (e) { downloadBlob(blob, name); return; }
    if (navigator.canShare && navigator.canShare({ files: [file] })) {
      navigator.share({ files: [file], title: name }).catch(function (err) {
        if (err && err.name === 'AbortError') return;
        downloadBlob(blob, name);
      });
    } else downloadBlob(blob, name);
  }
  function readFileText(f) {
    return new Promise(function (resolve, reject) {
      const fr = new FileReader();
      fr.onload = function () { resolve(String(fr.result)); };
      fr.onerror = function () { reject(new Error('File tidak bisa dibaca.')); };
      fr.readAsText(f);
    });
  }

  function saveHistory() {
    if (busy.save) return;
    const p = view.period;
    if (anyCfgErr()) { toast('Perbaiki input yang ditandai merah dulu.', 'warn'); return; }
    const inp = C.liveInputs(S, p);
    if (!(inp.annual > 0)) { toast('Isi annual salary dulu.', 'warn'); return; }
    const st = C.status(S, p);
    if (st === 'saved') { toast('Sudah tersimpan.', 'ok'); return; }
    busy.save = true;
    const go = st === 'changed'
      ? confirmModal('Timpa riwayat ' + C.periodLabel(p) + '?', 'Data tersimpan sebelumnya akan diganti dengan perhitungan saat ini.', 'Timpa')
      : Promise.resolve(true);
    go.then(function (ok) {

      const fresh = C.liveInputs(S, p);
      if (!(fresh.annual > 0)) return;
      S.history[p] = C.makeSnapshot(fresh, new Date().toISOString());
      S.months[p] = { annual: fresh.annual, insurance: fresh.insurance, pension: fresh.pension };
      if (persist()) toast('Tersimpan ke riwayat: ' + C.periodLabel(p), 'ok');
      if (view.period === p) { renderCfgMeta(); renderStatus(); }
    }).finally(function () { busy.save = false; });

  function sumCell(label, value, hl) { return '<div' + (hl ? ' class="hl"' : '') + '><small>' + label + '</small><b>' + value + '</b></div>'; }
  function itemHtml(it) {
    const r = it.res;
    const savedAt = new Date(it.snap.savedAt);
    const savedTxt = isNaN(savedAt.getTime()) ? '-' : savedAt.toLocaleString('id-ID', { day: '2-digit', month: 'short', year: 'numeric', hour: '2-digit', minute: '2-digit' });
    let det = '<div class="mini">' + lnRow('Annual salary', 'daily rate ' + C.fmtMoney(r.dr), C.fmtMoney(r.annual)) + slipRows(r);
    det += '<p>Tanggal ON: ' + (r.on ? C.dayRanges(r.onDates) : '-') + ' · Travel ON: ' + (r.ton ? C.dayRanges(r.tonDates) : '-') + ' · Travel OFF: ' + (r.toff ? C.dayRanges(r.toffDates) : '-') + '</p>';
    if (r.claims.length) det += '<p>Claim: ' + r.claims.map(function (c) { return esc(c.desc) + ' ' + C.fmtMoney(c.amount); }).join(' · ') + '</p>';
    det += '<p>Disimpan ' + esc(savedTxt) + '</p></div>';
    return '<article class="card h-item" data-period="' + r.period + '">' +
      '<div class="h-top"><div><h3>' + C.periodLabel(r.period) + '</h3><p>Dibayar ' + C.periodLabel(r.payPeriod) + '</p></div>' +
      '<div class="h-net">' + C.fmtMoney(r.net) + '<small>diterima</small></div></div>' +
      '<div class="h-chips"><span class="chip on">ON ' + r.on + '</span><span class="chip ton">Travel ON ' + r.ton + '</span><span class="chip toff">Travel OFF ' + r.toff + '</span><span class="chip off">OFF ' + r.off + '</span>' +
      (it.status === 'changed' ? '<span class="chip warn">Ada perubahan belum disimpan</span>' : '') + '</div>' +
      '<details><summary>Rincian</summary>' + det + '</details>' +
      '<div class="h-actions"><button type="button" class="btn" data-act="open">Buka</button><button type="button" class="btn ghost danger-text" data-act="del">Hapus</button></div></article>';
  }
  function renderRiwayat() {
    $('histGroup').value = view.histGroup;
    const years = C.historyYears(S, view.histGroup);
    if (!years.length) {
      $('yearChips').innerHTML = '';
      $('histSummary').hidden = true;
      $('histList').innerHTML = '<div class="empty">Belum ada riwayat.<br>Hitung bulannya di tab <b>Hitung</b>, lalu tekan <b>Simpan ke Riwayat</b>.</div>';
      return;
    }
    if (years.indexOf(view.histYear) < 0) view.histYear = years[0];
    $('yearChips').innerHTML = years.map(function (y) { return '<button type="button" class="chip-btn" data-year="' + y + '" aria-pressed="' + (y === view.histYear) + '">' + y + '</button>'; }).join('');
    const s = C.historySummary(S, view.histYear, view.histGroup);
    const sum = $('histSummary');
    sum.hidden = false;
    sum.innerHTML = '<div class="card-h"><h2>Ringkasan ' + view.histYear + '</h2><span class="hint">' + (view.histGroup === 'pay' ? 'per bulan dibayar' : 'per periode kerja') + ' · ' + s.count + ' bulan</span></div>' +
      '<div class="sum-grid">' + sumCell('Total diterima', C.fmtMoney(s.net), true) + sumCell('Total kotor', C.fmtMoney(s.gross)) +
      sumCell('Asuransi + pension', C.fmtMoney(s.deductions)) + sumCell('Claim expenses', C.fmtMoney(s.claims)) +
      sumCell('Hari ON', s.on + ' hari') + sumCell('ON + Travel', (s.on + s.trv) + ' / 182,5') +
      sumCell('Travel ON', s.ton + ' hari') + sumCell('Travel OFF', s.toff + ' hari') + '</div>';
    $('histList').innerHTML = s.items.map(itemHtml).join('');
  }
  function deleteHistory(p) {
    if (isModalOpen() || busy.del) return;
    if (!S.history[p]) return;
    busy.del = true;
    confirmModal('Hapus riwayat ' + C.periodLabel(p) + '?', 'Hanya riwayatnya yang dihapus. Tanggal di kalender dan claim bulan itu tetap ada.', 'Hapus', true).then(function (ok) {
      if (!ok || !S.history[p]) return;
      delete S.history[p];
      persist();
      renderRiwayat();
      toast('Riwayat ' + C.periodLabel(p) + ' dihapus.', 'ok');
    }).finally(function () { busy.del = false; });
  }

  /* ----- EXPORT ----- */
  function fillSelect(el, options, value) {
    el.innerHTML = options.map(function (o) { return '<option value="' + o[0] + '"' + (String(o[0]) === String(value) ? ' selected' : '') + '>' + o[1] + '</option>'; }).join('');
  }
  function renderExport() {
    const keys = Object.keys(S.history).sort();
    const cp = nowPeriod();
    if (!C.parsePeriod(view.exFrom) || !C.parsePeriod(view.exTo)) {
      view.exFrom = keys[0] || cp.slice(0, 4) + '-01';
      view.exTo = keys[keys.length - 1] || cp;
    }
    const ys = [+cp.slice(0, 4), +view.exFrom.slice(0, 4), +view.exTo.slice(0, 4)];
    if (keys.length) { ys.push(+keys[0].slice(0, 4)); ys.push(+keys[keys.length - 1].slice(0, 4)); }
    const minY = Math.max(C.LIMITS.yearMin, Math.min.apply(null, ys) - 1), maxY = Math.min(C.LIMITS.yearMax, Math.max.apply(null, ys) + 1);
    const yearOpts = [];
    for (let y = minY; y <= maxY; y++) yearOpts.push([y, y]);
    const monthOpts = C.MONTHS.map(function (m, i) { return [i + 1, m]; });
    fillSelect($('exFromM'), monthOpts, +view.exFrom.slice(5, 7));
    fillSelect($('exFromY'), yearOpts, +view.exFrom.slice(0, 4));
    fillSelect($('exToM'), monthOpts, +view.exTo.slice(5, 7));
    fillSelect($('exToY'), yearOpts, +view.exTo.slice(0, 4));
    document.querySelectorAll('input[name="exMode"]').forEach(function (r) { r.checked = r.value === view.exMode; });
    $('optDates').classList.toggle('sel', view.exMode === 'dates');
    $('optFull').classList.toggle('sel', view.exMode === 'full');
    $('btnExport').textContent = view.exMode === 'full' ? 'Download PDF (lengkap)' : 'Download PDF (tanggal)';
    renderExportInfo();
  }
  function readExportRange() {
    view.exFrom = C.pkey(+$('exFromY').value, +$('exFromM').value);
    view.exTo = C.pkey(+$('exToY').value, +$('exToM').value);
  }
  function shortList(ps) { return ps.slice(0, 12).map(C.periodShort).join(', ') + (ps.length > 12 ? ' dan ' + (ps.length - 12) + ' lainnya' : ''); }
  function renderExportInfo() {
    const from = view.exFrom, to = view.exTo;
    const info = $('exInfo'), btn = $('btnExport'), sh = $('btnShare');
    sh.hidden = !canShareFiles();
    if (!C.parsePeriod(from) || !C.parsePeriod(to) || from > to) {
      info.innerHTML = '<p class="banner warn">Bulan "Dari" harus sebelum atau sama dengan "Sampai".</p>';
      btn.disabled = true; sh.disabled = true;
      return;
    }
    const ps = C.periodsBetween(from, to);
    const saved = ps.filter(function (p) { return S.history[p]; });
    const missing = ps.filter(function (p) { return !S.history[p]; });
    const changed = saved.filter(function (p) { return C.status(S, p) === 'changed'; });
    let h;
    if (!saved.length) h = '<p class="banner warn">Belum ada riwayat tersimpan di rentang ini. Simpan dulu perhitungan bulanannya di tab Hitung.</p>';
    else {
      h = '<p><b>' + saved.length + '</b> dari ' + ps.length + ' bulan dalam rentang sudah tersimpan dan akan masuk PDF.</p>';
      if (missing.length) h += '<p class="tiny">Belum disimpan (tidak masuk PDF): ' + shortList(missing) + '</p>';
      if (changed.length) h += '<p class="banner warn">' + changed.length + ' bulan punya perubahan yang belum disimpan (' + shortList(changed) + '). PDF memakai versi yang tersimpan.</p>';
    }
    info.innerHTML = h;
    const dis = !saved.length || !!busy.export;
    btn.disabled = dis; sh.disabled = dis;
  }
  function doExport(share) {
    if (busy.export || locked('export', 1200)) return;
    readExportRange();
    const from = view.exFrom, to = view.exTo;
    if (!C.parsePeriod(from) || !C.parsePeriod(to) || from > to) { renderExportInfo(); return; }
    const ps = C.periodsBetween(from, to);
    const snaps = ps.filter(function (p) { return S.history[p]; }).map(function (p) { return S.history[p]; });
    const missing = ps.filter(function (p) { return !S.history[p]; });
    if (!snaps.length) { toast('Tidak ada riwayat tersimpan di rentang ini.', 'warn'); return; }
    busy.export = true;
    $('btnExport').disabled = true; $('btnShare').disabled = true;
    try {
      const bytes = Report.build(view.exMode, snaps, { from: from, to: to, missing: missing, now: new Date() });
      const name = Report.fileName(view.exMode, from, to);
      const blob = new Blob([bytes], { type: 'application/pdf' });
      if (share) shareOrDownload(blob, name);
      else { downloadBlob(blob, name); toast('PDF dibuat: ' + name, 'ok', 3500); }
    } catch (err) {
      if (window.console) console.error(err);
      toast('Gagal membuat PDF. ' + (err && err.message ? err.message : ''), 'err', 5000);
    } finally {
      setTimeout(function () { busy.export = false; renderExportInfo(); }, 900);
    }
  }

  /* ----- DATA ----- */
  function renderData() {
    $('appVersion').textContent = VERSION;
    let size = 0;
    try { size = (localStorage.getItem(KEY) || '').length; } catch (e) { size = 0; }
    $('storageSize').textContent = (size / 1024).toFixed(1) + ' KB';
  }
  function doBackup() {
    if (locked('backup', 1500)) return;
    const d = new Date();
    const payload = { app: 'gaji-rotasi', version: 1, exportedAt: d.toISOString(), data: S };
    const blob = new Blob([JSON.stringify(payload, null, 1)], { type: 'application/json' });
    downloadBlob(blob, 'gaji-rotasi-backup-' + C.ymd(d.getFullYear(), d.getMonth() + 1, d.getDate()) + '.json');
    toast('Backup dibuat.', 'ok');
  }
  function doRestore(f) {
    if (!f || busy.restore || isModalOpen()) return;
    busy.restore = true;
    Promise.resolve().then(function () {
      if (f.size > 5 * 1024 * 1024) throw new Error('File terlalu besar (maksimal 5 MB).');
      return readFileText(f);
    }).then(function (text) {
      let obj;
      try { obj = JSON.parse(text); } catch (e) { throw new Error('File bukan JSON yang valid.'); }
      if (!obj || obj.app !== 'gaji-rotasi' || typeof obj.data !== 'object' || obj.data === null) throw new Error('File ini bukan backup Gaji Rotasi.');
      const r = C.sanitizeState(obj.data);
      const nh = Object.keys(r.state.history).length, nd = Object.keys(r.state.days).length;
      return confirmModal('Pulihkan backup?', 'Backup berisi ' + nh + ' bulan riwayat dan ' + nd + ' tanggal ON/Travel' + (r.dropped ? ' (' + r.dropped + ' entri tidak valid akan diabaikan)' : '') + '. Semua data di perangkat ini akan diganti.', 'Pulihkan', true)
        .then(function (ok) {
          if (!ok) return;
          S = r.state;
          persist();
          view.period = S.ui.period || nowPeriod();
          view.brush = S.ui.brush || 'ON';
          view.exFrom = view.exTo = null; view.histYear = null;
          renderTab(view.tab);
          toast('Backup dipulihkan.', 'ok');
        });
    }).catch(function (err) { toast(err.message || 'Gagal memulihkan backup.', 'err', 5000); })
      .finally(function () { busy.restore = false; });
  }
  function wipe() {
    if (isModalOpen() || busy.wipe) return;
    busy.wipe = true;
    confirmModal('Hapus semua data?', 'Kalender, claim, dasar perhitungan, dan seluruh riwayat akan dihapus dari perangkat ini. Disarankan download backup dulu.', 'Hapus semua', true).then(function (ok) {
      if (!ok) return;
      S = C.emptyState();
      persist();
      view.period = nowPeriod(); view.brush = 'ON'; view.exFrom = view.exTo = null; view.histYear = null;
      renderTab(view.tab);
      toast('Semua data dihapus.', 'ok');
    }).finally(function () { busy.wipe = false; });
  }

  /* ----- tabs & init ----- */
  function renderTab(t) {
    if (t === 'hitung') renderHitung(true);
    else if (t === 'riwayat') renderRiwayat();
    else if (t === 'export') renderExport();
    else renderData();
  }
  function switchTab(t) {
    if (['hitung', 'riwayat', 'export', 'data'].indexOf(t) < 0) return;
    view.tab = t;
    document.querySelectorAll('section.tab').forEach(function (s) { s.hidden = s.dataset.tab !== t; });
    document.querySelectorAll('.tabbar button').forEach(function (b) { b.setAttribute('aria-selected', String(b.dataset.tab === t)); });
    renderTab(t);
    window.scrollTo(0, 0);
  }
  function bind() {
    document.querySelector('.tabbar').addEventListener('click', function (e) { const b = e.target.closest('button[data-tab]'); if (b) switchTab(b.dataset.tab); });
    $('prevMonth').addEventListener('click', function () { setPeriod(C.addMonths(view.period, -1)); });
    $('nextMonth').addEventListener('click', function () { setPeriod(C.addMonths(view.period, 1)); });
    $('goThisMonth').addEventListener('click', function () { setPeriod(nowPeriod()); });
    Object.keys(CFG).forEach(function (k) {
      const el = $(CFG[k].id);
      el.addEventListener('input', function () { onCfgInput(k); });
      el.addEventListener('focus', function () { onCfgFocus(k); });
      el.addEventListener('blur', function () { onCfgBlur(k); });
      el.addEventListener('keydown', function (e) { if (e.key === 'Enter') el.blur(); });
    });
    $('brushSeg').addEventListener('click', function (e) { const b = e.target.closest('button[data-brush]'); if (b) setBrush(b.dataset.brush); });
    $('calGrid').addEventListener('click', onCalClick);
    $('btnRange').addEventListener('click', openRange);
    $('btnClearMonth').addEventListener('click', clearMonth);
    $('btnAddClaim').addEventListener('click', addClaim);
    $('inClaimAmt').addEventListener('keydown', function (e) { if (e.key === 'Enter') { e.preventDefault(); addClaim(); } });
    $('inClaimDesc').addEventListener('keydown', function (e) { if (e.key === 'Enter') { e.preventDefault(); $('inClaimAmt').focus(); } });
    $('inClaimAmt').addEventListener('input', function () { $('errClaim').textContent = ''; });
    $('claimList').addEventListener('click', function (e) { const b = e.target.closest('button[data-id]'); if (b) deleteClaim(b.dataset.id); });
    $('btnSave').addEventListener('click', saveHistory);
    $('btnRevert').addEventListener('click', revert);
    $('yearChips').addEventListener('click', function (e) { const b = e.target.closest('button[data-year]'); if (!b) return; view.histYear = +b.dataset.year; renderRiwayat(); });
    $('histGroup').addEventListener('change', function () { view.histGroup = $('histGroup').value === 'pay' ? 'pay' : 'work'; view.histYear = null; renderRiwayat(); });
    $('histList').addEventListener('click', function (e) {
      const b = e.target.closest('button[data-act]');
      if (!b) return;
      const art = b.closest('article[data-period]');
      if (!art) return;
      const p = art.dataset.period;
      if (b.dataset.act === 'open') { if (!C.parsePeriod(p)) return; view.period = p; S.ui.period = p; persist(); switchTab('hitung'); }
      else if (b.dataset.act === 'del') deleteHistory(p);
    });
    ['exFromM', 'exFromY', 'exToM', 'exToY'].forEach(function (id) { $(id).addEventListener('change', function () { readExportRange(); renderExportInfo(); }); });
    $('exThisYear').addEventListener('click', function () { const y = nowPeriod().slice(0, 4); view.exFrom = y + '-01'; view.exTo = y + '-12'; renderExport(); });
    $('exAll').addEventListener('click', function () {
      const keys = Object.keys(S.history).sort();
      if (!keys.length) { toast('Belum ada riwayat.', 'warn'); return; }
      view.exFrom = keys[0]; view.exTo = keys[keys.length - 1]; renderExport();
    });
    document.querySelectorAll('input[name="exMode"]').forEach(function (r) { r.addEventListener('change', function () { if (r.checked) { view.exMode = r.value === 'full' ? 'full' : 'dates'; renderExport(); } }); });
    $('btnExport').addEventListener('click', function () { doExport(false); });
    $('btnShare').addEventListener('click', function () { doExport(true); });
    $('btnBackup').addEventListener('click', doBackup);
    $('inRestore').addEventListener('change', function (e) { const f = e.target.files && e.target.files[0]; e.target.value = ''; doRestore(f); });
    $('btnWipe').addEventListener('click', wipe);
    window.addEventListener('storage', function (e) {
      if (e.key !== KEY && e.key !== null) return;
      let raw = null;
      try { raw = localStorage.getItem(KEY); } catch (err) { return; }
      loadFromRaw(raw, true);
      if (!C.parsePeriod(view.period)) view.period = nowPeriod();
      if (!isModalOpen()) renderTab(view.tab);
    });
  }
  function registerSW() {
    if ('serviceWorker' in navigator && /^https?:$/.test(location.protocol)) {
      navigator.serviceWorker.register('sw.js').catch(function () { /* offline cache optional */ });
    }
  }
  function init() {
    load();
    view.period = S.ui.period || nowPeriod();
    view.brush = S.ui.brush || 'ON';
    bind();
    switchTab('hitung');
    registerSW();
  }
  return { init: init, version: VERSION, snapshot: function () { return JSON.parse(JSON.stringify(S)); } };
})();
window.GajiRotasi = App;
App.init();
